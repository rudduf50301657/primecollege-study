
# install.packages(ggplot2)
library(ggplot2)

# 종별 산점도
ggplot(iris, aes(x = Sepal.Length, y = Petal.Length)) + 
  geom_point(alpha = 0.7) + 
  facet_wrap(~ Species, nrow = 1)

# 두 변수 기준 격자 분할
# 데이터 전처리(임의 구간화) 포함
library(dplyr)
iris2 <- iris %>% 
  mutate(PetalWidthBin = cut(Petal.Width, 
                             breaks = c(0, 0.5, 1.5, Inf), 
                             labels = c("<=0.5", "0.5~1.5", ">1.5")))

ggplot(iris2, aes(Sepal.Length, Petal.Length)) + 
  geom_point() +
  facet_grid(Species ~ PetalWidthBin)

# 흰색 배경에 검은색 테두리가 있는 깔끔한 스타일 적용
ggplot(data = iris) + 
  geom_point(aes(x = Sepal.Length, y = Petal.Length, color = Species)) + 
  theme_bw()

# 불필요한 배경 요소(축 라인 등)를 제거한 현대적 스타일 적용
ggplot(data = iris) + 
  geom_point(aes(x = Sepal.Length, y = Petal.Length, color = Species)) + 
  theme_minimal()

# legend.position 옵션을 사용하여 범례를 아래쪽으로 이동
ggplot(data = iris) + 
  geom_point(aes(x = Sepal.Length, y = Petal.Length, color = Species)) + 
  theme(legend.position = "bottom")

# 축 이름의 크기, 색상, 글꼴 스타일 등을 직접 지정
ggplot(data = iris) + 
  geom_point(aes(x = Sepal.Length, y = Petal.Length, color = Species)) + 
  theme(axis.title = element_text(size = 14, color = "blue", face = "bold"))

# 종별 패널 분리
ggplot(iris, aes(x = Sepal.Length, y = Petal.Length)) +
  geom_point() +
  facet_wrap(~ Species)

# 회귀직선 추가
ggplot(iris, aes(x = Sepal.Length, y = Petal.Length)) +
  geom_point() +
  geom_smooth(method = "lm", se = FALSE)

# 밀도 그래프
ggplot(iris, aes(x = Sepal.Length, fill = Species)) +
  geom_density(alpha = 0.5)
